            <div class="workflow-step">
              <span class="step-number">2</span>
              <h3 class="h5">Validation and due diligence</h3>

            </div>
            <div class="workflow-step">
              <span class="step-number">3</span>
              <h3 class="h5">Review and approvals</h3>
              <p class="mb-0">Internal teams can review a complete application, request follow-up information, and record decisions in a consistent way.</p>
